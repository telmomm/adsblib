29static void
30_mutex_initialize(mutex_t* mutex) {
31#if defined(_WIN32)
32 InitializeCriticalSectionAndSpinCount(&mutex->csection, 4000);
33#else
34 pthread_mutexattr_t attr;
35 pthread_mutexattr_init(&attr);
36 pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_RECURSIVE);
37 pthread_mutex_init(&mutex->mutex, &attr);
38 pthread_mutexattr_destroy(&attr);
39#endif
40}

42static void
43_mutex_finalize(mutex_t* mutex) {

46#else

48#endif
